# pacman-explicit.txt — packages explicitly installed via pacman on Arch Linux
#
# These are packages that CANNOT or SHOULD NOT be managed by Nix Home Manager:
#   - require kernel/system integration (drivers, display server, init)
#   - only available in AUR / chaotic-aur
#   - provide systemd services that must run as root
#
# To reproduce on a new Arch install:
#   sudo pacman -S --needed $(grep -v '^#' pacman-explicit.txt | grep -v '^$')
#
# To audit what you've actually installed explicitly:
#   pacman -Qe | awk '{print $1}' | sort
